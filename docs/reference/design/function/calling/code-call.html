Fixed-order calls

An Agent method calls sub-methods in the order its source gives: survey, find gap, generate ideas; draft, review, revise; collect, clean, analyze.

from openprogram import Agent
from openprogram.runtime.agent import llm

class ExampleAgent(Agent):
    method_options = {'survey_pipeline': {'tool': True}}

    def survey_pipeline(self, task: str) -> str:
        survey = survey_topic(topic=task)
        if not survey or "error" in survey.lower():
            return "Survey failed"                     # check between steps

        key_points = extract_key_points(survey)        # plain Python between steps
        filtered = [p for p in key_points if p["relevance"] > 0.5]

        gaps = identify_gaps(survey="\n".join(filtered))  # a Python variable carries the data
        ideas = generate_ideas(gaps=gaps)

        return llm(f"Gaps:\n{gaps}\n\nIdeas:\n{ideas}")  # optional model call

_example_agent = ExampleAgent()
survey_pipeline = _example_agent.survey_pipeline
