class StuckToolWatcher(Policy):
    on = {"tool.after"}
    def evaluate(self, event, state):
        tool = event.payload["tool"]
        if state.tool_fail_count.get(tool, 0) >= 3:
            return Notify(f"{tool} has failed repeatedly, may be stuck")
